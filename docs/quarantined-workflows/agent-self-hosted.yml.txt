name: Run Agent on Self-Hosted Runner

on:
  push:
    branches: [ main ]
  workflow_dispatch:

jobs:
  agent:
    runs-on: [self-hosted, linux, x64, agent]
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: '3.11'

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

      - name: Run agent
        env:
          AGENT_API_KEY: ${{ secrets.AGENT_API_KEY }}
          AGENT_MODE: ${{ vars.AGENT_MODE || 'demo' }}
          AGENT_LOG_LEVEL: ${{ vars.AGENT_LOG_LEVEL || 'INFO' }}
        run: |
          python -u agents/run_task.py
